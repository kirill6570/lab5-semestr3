cmake_minimum_required(VERSION 3.16)
project(Lab5_ReportBuilder LANGUAGES CXX)
add_executable(lab5 main.cpp)
target_compile_features(lab5 PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(lab5 PRIVATE /W4 /utf-8)
else()
  target_compile_options(lab5 PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(builder_tests tests/test_builder.cpp)
  target_compile_features(builder_tests PRIVATE cxx_std_17)
  add_test(NAME builder_contract COMMAND builder_tests)
endif()
